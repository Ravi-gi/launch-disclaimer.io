/* SCROLL TO CHECKER */

    if (checker) {
        checker.scrollIntoView({
            behavior: "smooth",
            block: "start"

}


/* GET RADIO VALUE */

function getValue(name) {

    const selected = document.querySelector(
        'input[name="' + name + '"]:checked'
    );

    if (selected) {
        return selected.value;
    }

    return "no";
}


/* ESCAPE HTML */

function escapeHTML(text) {

    return String(text)
        .replace(/&/g, "&amp;")
        .replace(/</g, "&lt;")
        .replace(/>/g, "&gt;")
        .replace(/"/g, "&quot;")
        .replace(/'/g, "&#039;");
}


/* COPY TEXT - CHROME FRIENDLY */

function copyText(text) {

    if (navigator.clipboard && window.isSecureContext) {

        return navigator.clipboard.writeText(text);

    }

    /* Fallback for local HTML files */

    const textarea = document.createElement("textarea");

    textarea.value = text;

    textarea.style.position = "fixed";
    textarea.style.left = "-9999px";
    textarea.style.top = "0";

    document.body.appendChild(textarea);

    textarea.focus();
    textarea.select();

    let success = false;

    try {
        success = document.execCommand("copy");
    } catch (error) {
        success = false;
    }

    document.body.removeChild(textarea);

    if (success) {
        return Promise.resolve();
    }

    return Promise.reject(
        new Error("Copy operation failed")
    );
}


/* MAIN REPORT GENERATOR */

function generateReport() {

    const productInput =
        document.getElementById("product");

    const categoryInput =
        document.getElementById("category");


    const product =
        productInput.value.trim() || "Your Product";

    const category =
        categoryInput.value;


    if (!category) {

        alert("Please select a product category.");

        categoryInput.focus();

        return;
    }


    const privacy =
        getValue("privacy");

    const claims =
        getValue("claims");

    const children =
        getValue("children");


    let warnings = [];

    let score = 0;


    /* PRIVACY */

    if (privacy === "yes") {

        score += 20;

        warnings.push({

            code: "LD-PRIV-001",

            title: "Privacy Warning",

            message:
            "This product collects personal information. " +
            "Review applicable privacy requirements and " +
            "provide appropriate privacy disclosures.",

            level: "HIGH"

        });

    }


    /* ADVERTISING */

    if (claims === "yes") {

        score += 25;

        warnings.push({

            code: "LD-AD-001",

            title: "Advertising Claim Warning",

            message:
            "Health, safety or performance claims should " +
            "be supported by appropriate evidence before " +
            "being used in advertising or product packaging.",

            level: "HIGH"

        });

    }


    /* CHILDREN */

    if (children === "yes") {

        score += 20;

        warnings.push({

            code: "LD-CHILD-001",

            title: "Children's Product Warning",

            message:
            "This product is intended for children. " +
            "Additional child-safety, privacy and advertising " +
            "considerations may apply.",

            level: "MEDIUM"

        });

    }


    /* FOOD */

    if (category === "food") {

        score += 15;

        warnings.push({

            code: "LD-FOOD-001",

            title: "Food Product Warning",

            message:
            "Food and beverage products may require " +
            "appropriate labeling, safety information " +
            "and compliant advertising.",

            level: "MEDIUM"

        });

    }


    /* HEALTH */

    if (category === "health") {

        score += 20;

        warnings.push({

            code: "LD-HEALTH-001",

            title: "Health Product Warning",

            message:
            "Health-related products may require additional " +
            "safety, labeling, advertising and regulatory review.",

            level: "HIGH"

        });

    }


    /* FINANCE */

    if (category === "finance") {

        score += 25;

        warnings.push({

            code: "LD-FIN-001",

            title: "Financial Product Warning",

            message:
            "Financial products and services may be subject " +
            "to specific regulatory and consumer-protection requirements.",

            level: "HIGH"

        });

    }


    /* TECHNOLOGY */

    if (category === "technology") {

        warnings.push({

            code: "LD-TECH-001",

            title: "Technology Review",

            message:
            "Technology products may require review of " +
            "privacy, cybersecurity, intellectual property " +
            "and user-data practices.",

            level: "REVIEW"

        });

    }


    /* CHILDREN CATEGORY */

    if (category === "children" && children !== "yes") {

        score += 20;

        warnings.push({

            code: "LD-CHILD-002",

            title: "Children's Category Review",

            message:
            "The selected category indicates a children's " +
            "product. Review child-safety, privacy and " +
            "advertising considerations.",

            level: "MEDIUM"

        });

    }


    /* COPYRIGHT / IP */

    warnings.push({

        code: "LD-IP-001",

        title: "Intellectual Property Check",

        message:
        "Verify that logos, images, music, software, " +
        "trademarks and other third-party material " +
        "used by the product are properly authorized.",

        level: "REVIEW"

    });


    /* NEVER ALLOW SCORE ABOVE 100 */

    score = Math.min(score, 100);


    /* RISK LEVEL */

    let riskText;
    let riskColor;


    if (score >= 60) {

        riskText =
            "🔴 HIGH ATTENTION REQUIRED — " +
            score + "/100";

        riskColor = "#dc2626";

    } else if (score >= 30) {

        riskText =
            "🟡 REVIEW REQUIRED — " +
            score + "/100";

        riskColor = "#d97706";

    } else {

        riskText =
            "🟢 LOW PRELIMINARY RISK — " +
            score + "/100";

        riskColor = "#16a34a";

    }


    /* REPORT HEADER */

    document.getElementById("reportTitle").textContent =
        product + " — Legal Warning Report";

    const riskElement =
        document.getElementById("riskLevel");

    riskElement.textContent = riskText;
    riskElement.style.color = riskColor;


    /* WARNING CARDS */

    let html = "";


    warnings.forEach(function(warning) {

        let cardClass =
            warning.level === "HIGH"
            ? "warning-high"
            : warning.level === "MEDIUM"
            ? "warning-medium"
            : "warning-low";


        html += `
            <div class="warning-card ${cardClass}">

                <div class="warning-code">
                    ${escapeHTML(warning.code)}
                </div>

                <h3>
                    ${escapeHTML(warning.title)}
                </h3>

                <p>
                    ${escapeHTML(warning.message)}
                </p>

                <button
                    class="copy-btn"
                    type="button"
                    data-code="${escapeHTML(warning.code)}"
                >
                    Copy Warning Code
                </button>

            </div>
        `;

    });


    document.getElementById("warnings").innerHTML = html;


    /* ADD COPY BUTTON EVENTS */

    const copyButtons =
        document.querySelectorAll(".copy-btn");


    copyButtons.forEach(function(button) {

        button.addEventListener("click", function() {

            const code =
                button.getAttribute("data-code");

            copyText(code)
                .then(function() {

                    alert(
                        "Copied warning code: " + code
                    );

                })
                .catch(function() {

                    alert(
                        "Copy failed. Please copy the code manually: " +
                        code
                    );

                });

        });

    });


    /* SAVE CODES */

    window.generatedCodes =
        warnings
            .map(function(warning) {
                return warning.code;
            })
            .join("\n");


    /* SHOW RESULTS */

    const results =
        document.getElementById("results");

    results.style.display = "block";


    /* SCROLL TO RESULT */

    setTimeout(function() {

        results.scrollIntoView({
            behavior: "smooth",
            block: "start"
        });

    }, 100);

}


/* COPY ALL CODES */

function copyAllCodes() {

    const codes =
        window.generatedCodes || "";

    if (!codes) {

        alert("Generate a report first.");

        return;
    }


    copyText(codes)
        .then(function() {

            alert("All warning codes copied!");

        })
        .catch(function() {

            alert(
                "Copy failed. Please copy these codes manually:\n\n" +
                codes
            );

        });

}
